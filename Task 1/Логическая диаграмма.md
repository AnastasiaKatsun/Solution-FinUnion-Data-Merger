left to right direction

entity "Client" as CL {
  *global_client_id : UUID
  --
  full_name
  birth_date
  passport_data
  phone
  segment
  status
}

entity "Account" as AC {
  *account_id : UUID
  --
  global_client_id
  currency
  type
  status
  opened_at
  closed_at
  balance_ledger
  balance_available
}

entity "Contract" as CT {
  *contract_id : UUID
  --
  global_client_id
  account_id
  product_id
  signed_date
  status
  version
  document_ref
}

entity "Product" as PR {
  *product_id : UUID
  --
  name
  category
  tariff_rules
  valid_from
  valid_to
  is_active
}

entity "Transaction" as TX {
  *tx_id : UUID
  --
  account_id
  card_id
  amount
  currency
  tx_type
  timestamp_utc
  status
  source_system
}

entity "Case" as CS {
  *case_id : UUID
  --
  global_client_id
  channel
  category
  priority
  status
  created_at
  resolved_at
}

' Связи
CL }|--o{ AC : имеет (1:N)
CL }|--o{ CT : заключает (1:N)
CL }|--o{ CS : создаёт (1:N)
AC }|--o{ TX : содержит операции (1:N)
CT ||--|| PR : ссылается на продукт (N:1)
AC ||--o{ CT : счёт связан с договором (1:N)
TX }|--|| AC : относится к счёту (N:1)

' Опционально, чтобы схема выглядела «по-банковски»
entity "Card" as CD {
  *card_id : UUID
  --
  account_id
  pan_last4
  status
  issued_at
  expiry_date
}
CD }|--|| AC : привязана к счёту (1:1 или 1:N)
TX }|--o{ CD : может быть по карте (0:N)

note right of AC
  Account — мастер-сущность в Core Banking.
  Один клиент может иметь много счетов.
end note

note bottom of CT
  Contract — юридически значимая сущность.
  Договор ссылается на продукт и счёт.
end note

note left of TX
  Transaction — основа для отчётности и антифрода.
  Все операции привязаны к счёту.
end note
